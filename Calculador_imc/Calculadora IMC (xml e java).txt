package com.example.calculadorimc;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class MainActivity extends AppCompatActivity{

    EditText valorPeso, valorAltura;
    TextView campo_resultadoIMC, campo_classificacao;

    float peso, altura, valorIMC;

    @Override
    protected void onCreate(Bundle savedInstanceState){
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        valorPeso = findViewById(R.id.edt_peso);
        valorAltura = findViewById(R.id.edt_altura);
        campo_resultadoIMC = findViewById(R.id.txt_valorimc);
        campo_classificacao = findViewById(R.id.txt_classificacao);
        Button calcularIMC = findViewById(R.id.btn_calcularimc);

        calcularIMC.setOnClickListener(new View.OnClickListener(){
            @Override
            public void onClick(View view) {
                calcular_imc();
            }
        });
    }

    public void calcular_imc(){
        String texto1 = valorPeso.getText().toString().trim();
        String texto2 = valorAltura.getText().toString().trim();

        if(texto1.isEmpty()){
            valorPeso.setError("Digite o peso");
            return;
        }
        if(texto2.isEmpty()){
            valorAltura.setError("Digite a altura");
            return;
        }

        try{
            peso = Float.parseFloat(valorPeso.getText().toString());
            altura = Float.parseFloat(valorAltura.getText().toString());
        }
        catch(NumberFormatException e){
            campo_resultadoIMC.setText("Número Inválido");
            return;
        }

        if(peso <= 0 || altura <= 0){
            campo_resultadoIMC.setText("Digite valores maiores que zero");
            return;
        }

        valorIMC = peso / (altura * altura);

        campo_resultadoIMC.setText("IMC: " + String.format("%.2f", valorIMC));


        if(valorIMC < 18.5){
            campo_classificacao.setText("Baixo peso");
        }
        else if(valorIMC >= 18.6 && valorIMC <= 24.9){
            campo_classificacao.setText("Peso Normal");
        }
        else if(valorIMC >= 25 && valorIMC <= 29.9){
            campo_classificacao.setText("Sobrepeso");
        }
        else if(valorIMC >= 30 && valorIMC <= 34.9){
            campo_classificacao.setText("Obesidade grau I");
        }
        else if(valorIMC >= 35 && valorIMC <=39.9){
            campo_classificacao.setText("Obesidade grau II");
        }
        else{
            campo_classificacao.setText("Obesidade grau III");
        }
    }
}
